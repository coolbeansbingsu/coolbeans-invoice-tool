// Cool Beans Bingsu — Invoice Logger + Anthropic Proxy // Paste this into Extensions > Apps Script in your Google Sheet var ANTHROPIC_API_KEY = "YOUR_ANTHROPIC_API_KEY_HERE"; // paste your Anthropic key here function doPost(e) { try { const data = JSON.parse(e.postData.contents); if (data.action === "parse") { return handleParse(data.inquiryText); } else if (data.action === "log") { return handleLog(data); } else { throw new Error("Unknown action"); } } catch(err) { return ContentService .createTextOutput(JSON.stringify({ status: "error", message: err.message })) .setMimeType(ContentService.MimeType.JSON); } } function handleParse(inquiryText) { const prompt = "Extract the following fields from this catering inquiry form submission and return ONLY a JSON object with no markdown or extra text:\n{\n \"clientName\": \"\",\n \"clientEmail\": \"\",\n \"eventDate\": \"YYYY-MM-DD format\",\n \"eventType\": \"\",\n \"venue\": \"\",\n \"startTime\": \"HH:MM in 24hr format\",\n \"endTime\": \"HH:MM in 24hr format\",\n \"servings\": number,\n \"guestCount\": number,\n \"setupSpace\": \"indoor OR outdoor_shaded OR outdoor_canopy\",\n \"additionalNotes\": \"\"\n}\n\nFor setupSpace: if indoor use indoor, if outdoor with no canopy needed use outdoor_shaded, if canopy needed use outdoor_canopy.\nFor times: parse from additional notes if not explicitly given (e.g. 1-2:30pm means startTime 13:00, endTime 14:30).\n\nInquiry:\n" + inquiryText; const payload = { model: "claude-sonnet-4-20250514", max_tokens: 1000, messages: [{ role: "user", content: prompt }] }; const options = { method: "post", contentType: "application/json", headers: { "x-api-key": ANTHROPIC_API_KEY, "anthropic-version": "2023-06-01" }, payload: JSON.stringify(payload), muteHttpExceptions: true }; const response = UrlFetchApp.fetch("https://api.anthropic.com/v1/messages", options); const result = JSON.parse(response.getContentText()); if (result.error) throw new Error(result.error.message); const text = result.content.map(function(i) { return i.text || ""; }).join(""); const clean = text.replace(/```json|```/g, "").trim(); return ContentService .createTextOutput(JSON.stringify({ status: "success", parsed: JSON.parse(clean) })) .setMimeType(ContentService.MimeType.JSON); } function handleLog(data) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); if (sheet.getLastRow() === 0) { sheet.appendRow([ "Invoice #", "Date Issued", "Client Name", "Email", "Event Date", "Event Type", "Venue", "Service Time", "Servings", "Guest Count", "Mileage (mi)", "Base Fee", "Bingsu Total", "Mileage Fee", "Add. Service Time", "Gratuity", "Canopy", "Grand Total", "Deposit", "Final Payment", "Final Due Date" ]); } sheet.appendRow([ data.invoiceNum || "", data.dateIssued || "", data.clientName || "", data.clientEmail || "", data.eventDate || "", data.eventType || "", data.venue || "", data.serviceTime || "", data.servings || 0, data.guestCount || 0, data.mileage || 0, data.baseFee || 0, data.bingsuTotal || 0, data.mileageTotal || 0, data.addTimeTotal || 0, data.gratuityTotal || 0, data.canopyTotal || 0, data.grandTotal || 0, data.deposit || 0, data.finalPay || data.deposit || 0, data.finalDue || "Due At Signing" ]); return ContentService .createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); } function doGet(e) { return ContentService .createTextOutput(JSON.stringify({ status: "ok" })) .setMimeType(ContentService.MimeType.JSON); }
